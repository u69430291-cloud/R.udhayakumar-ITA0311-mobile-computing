        "%%writefile exp5.tcl\n",
        "\n",
        "set ns [new Simulator]\n",
        "\n",
        "# Enable Distance Vector routing\n",
        "$ns rtproto DV\n",
        "\n",
        "set tr [open out.tr w]\n",
        "set nf [open out.nam w]\n",
        "\n",
        "$ns trace-all $tr\n",
        "$ns namtrace-all $nf\n",
        "\n",
        "# Create four nodes\n",
        "for {set i 0} {$i < 4} {incr i} {\n",
        "    set n($i) [$ns node]\n",
        "}\n",
        "\n",
        "# Create ring topology\n",
        "for {set i 0} {$i < 4} {incr i} {\n",
        "    $ns duplex-link $n($i) $n([expr ($i+1)%4]) 1Mb 10ms DropTail\n",
        "}\n",
        "\n",
        "# UDP connection from node 0 to node 2\n",
        "set udp [new Agent/UDP]\n",
        "set null [new Agent/Null]\n",
        "\n",
        "$ns attach-agent $n(0) $udp\n",
        "$ns attach-agent $n(2) $null\n",
        "$ns connect $udp $null\n",
        "\n",
        "# CBR traffic\n",
        "set cbr [new Application/Traffic/CBR]\n",
        "$cbr attach-agent $udp\n",
        "$cbr set packetSize_ 512\n",
        "$cbr set interval_ 0.05\n",
        "\n",
        "# Start traffic\n",
        "$ns at 0.5 \"$cbr start\"\n",
        "\n",
        "# Link failure and recovery\n",
        "$ns rtmodel-at 2.0 down $n(1) $n(2)\n",
        "$ns rtmodel-at 3.0 up $n(1) $n(2)\n",
        "\n",
        "# Stop traffic and finish\n",
        "$ns at 4.5 \"$cbr stop\"\n",
        "$ns at 5.0 \"finish\"\n",
        "\n",
        "proc finish {} {\n",
        "    global ns tr nf\n",
        "    $ns flush-trace\n",
        "    close $tr\n",
        "    close $nf\n",
        "    exit 0\n",
        "}\n",
        "\n",
        "$ns run"
