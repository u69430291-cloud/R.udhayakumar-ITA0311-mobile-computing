        "%%writefile exp2.tcl\n",
        "\n",
        "set ns [new Simulator]\n",
        "\n",
        "set tr [open out.tr w]\n",
        "set nf [open out.nam w]\n",
        "\n",
        "$ns trace-all $tr\n",
        "$ns namtrace-all $nf\n",
        "\n",
        "# Create three nodes\n",
        "set n0 [$ns node]\n",
        "set n1 [$ns node]\n",
        "set n2 [$ns node]\n",
        "\n",
        "# Connect nodes\n",
        "$ns duplex-link $n0 $n1 2Mb 10ms DropTail\n",
        "$ns duplex-link $n1 $n2 1Mb 20ms DropTail\n",
        "\n",
        "# TCP agent\n",
        "set tcp [new Agent/TCP]\n",
        "set sink [new Agent/TCPSink]\n",
        "\n",
        "$ns attach-agent $n0 $tcp\n",
        "$ns attach-agent $n2 $sink\n",
        "$ns connect $tcp $sink\n",
        "\n",
        "# FTP application\n",
        "set ftp [new Application/FTP]\n",
        "$ftp attach-agent $tcp\n",
        "\n",
        "# Start and stop FTP\n",
        "$ns at 0.5 \"$ftp start\"\n",
        "$ns at 4.5 \"$ftp stop\"\n",
        "\n",
        "# Finish simulation\n",
        "$ns at 5.0 \"finish\"\n",
        "\n",
        "proc finish {} {\n",
        "    global ns tr nf\n",
        "    $ns flush-trace\n",
        "    close $tr\n",
        "    close $nf\n",
        "    exit 0\n",
        "}\n",
        "\n",
        "$ns run"
