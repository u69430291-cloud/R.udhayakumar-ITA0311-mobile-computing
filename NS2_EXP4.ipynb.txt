        "%%writefile exp4.tcl\n",
        "\n",
        "set ns [new Simulator]\n",
        "\n",
        "set tr [open out.tr w]\n",
        "set nf [open out.nam w]\n",
        "\n",
        "$ns trace-all $tr\n",
        "$ns namtrace-all $nf\n",
        "\n",
        "# Create four nodes\n",
        "set n0 [$ns node]\n",
        "set n1 [$ns node]\n",
        "set n2 [$ns node]\n",
        "set n3 [$ns node]\n",
        "\n",
        "# Create Ethernet LAN\n",
        "$ns make-lan \"$n0 $n1 $n2 $n3\" 10Mb 10ms LL Queue/DropTail Mac/802_3\n",
        "\n",
        "# TCP connection\n",
        "set tcp [$ns create-connection TCP $n0 TCPSink $n3 0]\n",
        "\n",
        "# FTP application\n",
        "set ftp [new Application/FTP]\n",
        "$ftp attach-agent $tcp\n",
        "\n",
        "# Start and stop FTP\n",
        "$ns at 0.5 \"$ftp start\"\n",
        "$ns at 4.5 \"$ftp stop\"\n",
        "\n",
        "# Finish simulation\n",
        "$ns at 5.0 \"finish\"\n",
        "\n",
        "proc finish {} {\n",
        "    global ns tr nf\n",
        "    $ns flush-trace\n",
        "    close $tr\n",
        "    close $nf\n",
        "    exit 0\n",
        "}\n",
        "\n",
        "$ns run"
